 create table userProfile (
   profile_id INT AUTO_INCREMENT PRIMARY KEY,
   user_id VARCHAR(50),
   profile_image VARCHAR(255),
   FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE
 );
 
 create table usernotification(
   id INT AUTO_INCREMENT PRIMARY KEY,            -- 기본 키(id)
    user_id VARCHAR(30),                          -- user 테이블의 user_id를 참조하는 외래 키
    fcm_token VARCHAR(255),                       -- FCM 토큰
    CONSTRAINT fk_user_id FOREIGN KEY (user_id)    -- 외래 키 제약 조건
    REFERENCES user(user_id) ON DELETE CASCADE  
);

create table userHelpCall(
   id int auto_increment primary key,
    user_id varchar(30),
    latitude varchar(50),
    longitude varchar(50),
    constraint fk_userHelpCall_user_id foreign key (user_id)
    references user(user_id) on delete cascade
);

CREATE TABLE complaint (

    complaint_content    VARCHAR(700),
    complaint_type        VARCHAR(10),
    complaint_state      VARCHAR(4) DEFAULT '미처리',

    updated_at           TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,          
    complaint_img        VARCHAR(300),
    user_id              VARCHAR(50),
    mountain_name        VARCHAR(50),
    latitude             double, 
    longitude            double,
    institution         varchar(100)
);

AFTER INSERT ON processing
FOR EACH ROW
BEGIN

CREATE TABLE course_reviews (
    review_id INT AUTO_INCREMENT PRIMARY KEY,
    course_id INT NOT NULL,  
    user_id VARCHAR(50) NOT NULL,  
    review_content TEXT, 
    rating INT,  -- 별점 (1~5)
    image_path VARCHAR(255),
    difficulty VARCHAR(50), -- 난이도
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (course_id) REFERENCES course(course_id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE
);

# 코스 평점 course_rate 업데이트하는 트리거 
DELIMITER //

CREATE TRIGGER update_course_rate_after_insert
AFTER INSERT ON course_reviews
FOR EACH ROW
BEGIN
    DECLARE avg_rating DECIMAL(2, 1);
    
    -- 현재 course_id에 대한 rating 평균을 계산
    SELECT AVG(rating)
    INTO avg_rating
    FROM course_reviews
    WHERE course_id = NEW.course_id;

    -- course 테이블의 course_rate를 업데이트
    UPDATE course
    SET course_rate = avg_rating
    WHERE course_id = NEW.course_id;
END //

DELIMITER ;
# 코스 난이도 course_level 업데이트하는 트리거 
DELIMITER //

CREATE TRIGGER update_course_level_after_insert
AFTER INSERT ON course_reviews
FOR EACH ROW
BEGIN
    DECLARE avg_difficulty DECIMAL(2, 1);
    
    -- difficulty 값이 문자열로 저장되어 있으면, 이를 숫자로 변환하여 평균을 계산
    SELECT AVG(
        CASE 
            WHEN difficulty = '쉬움' THEN 1
            WHEN difficulty = '보통' THEN 2
            WHEN difficulty = '어려움' THEN 3
            ELSE 0
        END
    ) INTO avg_difficulty
    FROM course_reviews
    WHERE course_id = NEW.course_id;

    -- course 테이블의 course_difficulty를 업데이트
    UPDATE course
    SET course_level = 
        CASE
            WHEN avg_difficulty <= 1.3 THEN '쉬움'
            WHEN avg_difficulty <= 2.3 THEN '보통'
            ELSE '어려움'
        END
    WHERE course_id = NEW.course_id;
END //

DELIMITER ;

CREATE TABLE path_info (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id varchar(50), -- 사용자 ID
    start_time DATETIME, -- 경로 시작 시간
    end_time DATETIME, -- 경로 종료 시간
    total_distance DECIMAL(10, 2), -- 총 거리 (옵션)
    path_img varchar(300),
    total_time varchar(8),
    max_altitude decimal(10,7),
    avg_speed float,
    FOREIGN KEY (user_id) REFERENCES user(user_id) -- users 테이블과 연결
);

CREATE TABLE tracking_path (
    id INT AUTO_INCREMENT PRIMARY KEY,
    path_id INT NOT NULL, -- 경로를 식별하기 위한 ID
    latitude DECIMAL(10, 7) NOT NULL, -- 위도
    longitude DECIMAL(10, 7) NOT NULL, -- 경도
    altitude decimal(10,7) NOT NULL,
    speed float NOT NULL,
    bearing float NOT NULL,
    timelog varchar(20), -- 위치 기록 시간
    FOREIGN KEY (path_id) REFERENCES path_info(id) ON DELETE CASCADE -- path_info와 연결
);   


DELIMITER //

CREATE TRIGGER update_path_info_stats
AFTER INSERT ON tracking_path
FOR EACH ROW
BEGIN
    DECLARE new_total_distance FLOAT;

    -- 거리 계산
    WITH distance_calculations AS (
        SELECT 
            path_id,
            latitude,
            longitude,
            LEAD(latitude) OVER (PARTITION BY path_id ORDER BY id) AS next_latitude,
            LEAD(longitude) OVER (PARTITION BY path_id ORDER BY id) AS next_longitude
        FROM tracking_path
    ),
    distances AS (
        SELECT
            path_id,
            6371000 * 2 * ASIN(SQRT(
                POWER(SIN(RADIANS(next_latitude - latitude) / 2), 2) +
                COS(RADIANS(latitude)) * COS(RADIANS(next_latitude)) *
                POWER(SIN(RADIANS(next_longitude - longitude) / 2), 2)
            )) / 1000 AS segment_distance -- 미터를 킬로미터로 변환
        FROM distance_calculations
        WHERE next_latitude IS NOT NULL AND next_longitude IS NOT NULL
    )
    SELECT 
        SUM(segment_distance) INTO new_total_distance
    FROM distances
    WHERE path_id = NEW.path_id;

    -- avg_speed와 max_altitude 계산 후 path_info에 업데이트
    UPDATE path_info
    SET 
        avg_speed = (SELECT AVG(speed) FROM tracking_path WHERE path_id = NEW.path_id),
        max_altitude = (SELECT MAX(altitude) FROM tracking_path WHERE path_id = NEW.path_id),
        total_distance = new_total_distance
    WHERE id = NEW.path_id;

END //

DELIMITER ;
